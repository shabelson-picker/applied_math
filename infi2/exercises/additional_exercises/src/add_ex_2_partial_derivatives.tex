\section*{Exercise 1: Find the first-order partial derivatives}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]
\item \easy\quad $z = x^3 + 3x^2 y - y^3$
\item \easy\quad $z = \ln(x^2+y^2)$
\item \easy\quad $z = \arctan\!\left(\dfrac{y}{x}\right)$
\item \easy\quad $z = \cos(ax-by)$
\item \medium\quad $u = xe^{-yx}$
\item \medium\quad $u = \dfrac{y}{x}+\dfrac{z}{y}+\dfrac{x}{z}$
\item \medium\quad $z = \dfrac{xy}{x-y}$
\item \medium\quad $u = \arcsin\!\left(\dfrac{y}{x}\right)$
\item \hard\quad $u = \sin^2(x+y) - \sin^2 x - \sin^2 y$
\item \hard\quad $z = \ln\!\left(\sqrt[3]{\dfrac{1}{x}}-\sqrt[3]{\dfrac{1}{y}}\right)$
\end{enumerate}

\section*{Exercise 2: Calculate the partial derivatives at $(0,0)$ from the definition}
%----------------------------------------------------------

\emph{Do not use the standard differentiation rules --- use the limit definition directly.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
$f(x,y)=\begin{cases}
\dfrac{x^3+y^3}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{2xy}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{xy}{\sqrt{x^2+y^2}}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
x^2\sin\!\dfrac{1}{y}+y^2\sin\!\dfrac{1}{x}, & x\ne 0,\; y\ne 0,\\[4pt]
0, & \text{otherwise}.
\end{cases}$

\item \hard\quad
$f(x,y)=\begin{cases}
\dfrac{x^3 y - xy^3}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\end{enumerate}

\section*{Exercise 3: Verify the given identity}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \medium\quad
$z = \ln(\sqrt{x}+\sqrt{y})$.\quad
Verify: $x\,\dfrac{\partial z}{\partial x}+y\,\dfrac{\partial z}{\partial y}=\dfrac{1}{2}$.

\item \medium\quad
$u = e^{x/t^2}$.\quad
Verify: $2x\,\dfrac{\partial u}{\partial x}+t\,\dfrac{\partial u}{\partial t}=0$.

\item \medium\quad
$u = f(x^2+y^2+z^2)$, $f$ differentiable. Verify:
\[y\,\frac{\partial u}{\partial x}-x\,\frac{\partial u}{\partial y}=0.\]

\item \hard\quad
$u = \sqrt{x^2+y^2+z^2}$.\quad
Verify: $\left(\dfrac{\partial u}{\partial x}\right)^2+\left(\dfrac{\partial u}{\partial y}\right)^2+\left(\dfrac{\partial u}{\partial z}\right)^2 = 1$.

\item \hard\quad
$u = f\!\left(\dfrac{x}{y},\dfrac{y}{z}\right)$, $f$ differentiable. Show that
$x\,\dfrac{\partial u}{\partial x}+y\,\dfrac{\partial u}{\partial y}+z\,\dfrac{\partial u}{\partial z}=0$.

\end{enumerate}

\section*{Exercise 4: Check differentiability at $(0,0)$}
%----------------------------------------------------------

\emph{For each function: (1) compute $f_x(0,0)$, $f_y(0,0)$ from the limit definition;
(2) check whether $\lim_{\Delta\rho\to 0}[\Delta f - f_x\Delta x - f_y\Delta y]/\Delta\rho = 0$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{x^3+y^3}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \hard\quad
$f(x,y)=\begin{cases}
(x^2+y^2)\cos\!\dfrac{\pi}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \hard\quad
$f(x,y) = \sqrt[3]{xy}$\quad at $(0,0)$.\\[4pt]
(a) Find $f_x(0,0)$ and $f_y(0,0)$.\\
(b) Are $f_x,f_y$ continuous at $(0,0)$?\\
(c) Is $f$ differentiable at $(0,0)$?\\
\emph{(Hint: (b) and (c) have different answers.)}

\item \hard\quad
$f(x,y)=\begin{cases}
\dfrac{x^2 y}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$\\[4pt]
\emph{Are the partial derivatives continuous at $(0,0)$? Is $f$ differentiable?}

\end{enumerate}

\section*{Exercise 5: Tangent plane and normal line}
%----------------------------------------------------------

\emph{Recall: tangent plane to $z=f(x,y)$ at $(a,b)$:
$z - f(a,b) = f_x(a,b)(x-a)+f_y(a,b)(y-b)$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
$z = x^2+y^2$ at the point $(1,1,2)$.

\item \medium\quad
$z = e^x\sin y$ at the point $(0,\pi/2,1)$.

\item \medium\quad
$z = \ln(x^2+y^2)$ at the point $(1,0,0)$.

\item \medium\quad
$z = \sqrt{x^2+y^2}$ at the point $(3,4,5)$.

\item \hard\quad
$x^2+y^2+z^2 = 14$ at the point $(1,2,3)$.\\
\emph{(Level surface form --- use $\nabla F$.)}

\item \hard\quad
Find the points on the surface $z = x^2 - xy + y^2 + 1$ where the tangent plane
is parallel to the plane $2x + 4y - z = 5$.

\end{enumerate}
