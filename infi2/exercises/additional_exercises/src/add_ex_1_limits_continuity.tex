\section*{Exercise 1: Find and sketch the domains of the following functions}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]
\item \easy\quad $f(x,y) = \ln(x^2 - y)$
\item \easy\quad $f(x,y) = \sqrt{9 - x^2 - y^2}$
\item \medium\quad $f(x,y) = \arcsin\!\left(\dfrac{x+y}{x-y}\right)$
\item \medium\quad $f(x,y) = \ln\!\left(\dfrac{x^2+y^2}{x^2-y^2}\right)$
\item \hard\quad $f(x,y) = \sqrt{\ln(x^2+y^2-1)}$
\end{enumerate}

\section*{Exercise 3: Find the limits or show that they do not exist}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{\sin(x^2+y^2)}{x^2+y^2}$

\item \easy\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{2x^2+3y^4}{x^2+y^2}$

\item \easy\quad
$\displaystyle\lim_{(x,y)\to(0,0)} (x^2+y^2)\cos\!\left(\frac{1}{xy}\right)$

\item \medium\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{x^3 y}{x^4+y^2}$

\item \medium\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{x^2+y^2}{\sqrt{x^2+y^2+1}-1}$

\item \medium\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{\sin(x^2+y^4)}{x^2+y^2}$

\item \medium\quad
$\displaystyle\lim_{\substack{x\to\infty\\y\to 3}}\left(1+\frac{1}{x}\right)^{x+y}$

\item \hard\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{\sin(xy)}{xy}\cdot\sin\!\left(\frac{1}{\sqrt{x^2+y^2}}\right)$

\item \hard\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{x^5 - x^2 y^3}{x^4+y^4}$

\item \hard\quad
$\displaystyle\lim_{(x,y)\to(0,0)} \frac{x^5 - \cos(x^2+y^2)+1}{(x^2+y^2)^2}$

\end{enumerate}

\section*{Exercise 4: Find the maximal domain on which each function is continuous}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
$f(x,y)=\begin{cases}
\dfrac{x^2+2y^2}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
1, & (x,y)=(0,0).
\end{cases}$

\item \easy\quad
$f(x,y)=\begin{cases}
\dfrac{x^2 y^2}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{\sqrt[3]{y}\,\sin(x^2)}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
(x+y)\sin\!\dfrac{1}{x^2}\cos\!\dfrac{1}{y}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{x^3+y^3}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \medium\quad
$f(x,y)=\begin{cases}
\dfrac{x^4 y^2}{x^4+y^2}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \hard\quad
$f(x,y)=\begin{cases}
\dfrac{x^4-x^3 y-xy^3+y^4}{\sqrt{x^6+y^6}}, & (x,y)\ne(0,0),\\[6pt]
0, & (x,y)=(0,0).
\end{cases}$

\item \hard\quad
$f(x,y)=\begin{cases}
(x^2+y^2)\sin\!\dfrac{1}{x^2+y^2}, & (x,y)\ne(0,0),\\[6pt]
1, & (x,y)=(0,0).
\end{cases}$

\end{enumerate}
