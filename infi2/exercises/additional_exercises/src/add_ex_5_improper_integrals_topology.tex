\section*{Exercise 1: Determine convergence or divergence}
%----------------------------------------------------------

\emph{Identify the type (I or II). Find the asymptotic behaviour near the singular point.
Compare with the reference $p$-integral.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $\displaystyle\int_1^{\infty}\frac{dx}{x^3+1}$

\item \easy\quad $\displaystyle\int_0^{1}\frac{dx}{\sqrt[3]{x}}$

\item \easy\quad $\displaystyle\int_1^{\infty}\frac{\ln x}{x^2}\,dx$

\item \easy\quad $\displaystyle\int_0^{1}\frac{\sin x}{\sqrt{x}}\,dx$

\item \easy\quad $\displaystyle\int_0^{\infty} e^{-x}\,dx$

\item \easy\quad $\displaystyle\int_2^{\infty}\frac{dx}{x(\ln x)^2}$

\item \medium\quad $\displaystyle\int_1^{\infty}\!\bigl(\sqrt{x+1}-\sqrt{x-1}\bigr)\,dx$

\item \medium\quad $\displaystyle\int_0^{1}\frac{\ln(1+\sqrt[3]{x})}{e^{\sin x}-1}\,dx$

\item \medium\quad $\displaystyle\int_0^{\infty}\frac{\sin x}{x^{3/2}}\,dx$
\quad\emph{(treat $x\to 0^+$ and $x\to\infty$ separately)}

\item \medium\quad $\displaystyle\int_0^{1}\frac{e^x - 1}{\sin^2 x}\,dx$

\item \medium\quad $\displaystyle\int_1^{\infty}\frac{\arctan x}{x^2}\,dx$

\item \hard\quad $\displaystyle\int_1^{\infty}\!\left(\sqrt{x^2+1}-\sqrt{x^2-1}\right)dx$

\item \hard\quad $\displaystyle\int_0^{1}\frac{\sqrt[3]{x}\ln x}{1-x^2}\,dx$

\end{enumerate}

\section*{Exercise 2: Topology of $\mathbb{R}^n$ --- classify each set}
%----------------------------------------------------------

\emph{For each set, identify:
(a) whether a given sample point is interior, boundary, or exterior;
(b) whether the set is open, closed, both, or neither;
(c) whether the set is bounded;
(d) whether the set is compact.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $E = \{(x,y) : x^2+y^2 < 1\}$

\item \easy\quad $E = \{(x,y) : x^2+y^2 \le 1\}$

\item \easy\quad $E = \{(x,y) : x^2+y^2 = 1\}$ \quad (the circle itself)

\item \easy\quad $E = \{(x,y,z) : x^2+y^2+z^2 \le 4\}$

\item \medium\quad $E = \{(x,y) : 1 < x^2+y^2 \le 4\}$

\item \medium\quad $E = \{(x,y) : y > x^2\}$

\item \medium\quad $E = \{(x,y) : x > 0,\; y > 0,\; xy \le 1\}$

\item \medium\quad $E = \{(x,y) : |x|+|y| \le 1\}$

\item \hard\quad $E = \mathbb{Q}^2 = \{(x,y) : x\in\mathbb{Q},\; y\in\mathbb{Q}\}$

\item \hard\quad $E = \{(x,y) : y = \sin(1/x),\; x > 0\}$

\end{enumerate}

\section*{Exercise 3: Norms and inequalities}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
Compute $\|\bar{x}\|_1$, $\|\bar{x}\|_2$, $\|\bar{x}\|_\infty$ for $\bar{x}=(3,-4,0)$.

\item \easy\quad
Compute $\|\bar{x}\|_1$, $\|\bar{x}\|_2$, $\|\bar{x}\|_\infty$ for $\bar{x}=(1,2,-2,0)$.

\item \medium\quad
Show that for any $\bar{x}\in\mathbb{R}^2$:
$\|\bar{x}\|_\infty \le \|\bar{x}\|_2 \le \|\bar{x}\|_1$.

\item \medium\quad
Verify the Cauchy--Schwarz inequality for
$\bar{x}=(1,2,3)$, $\bar{y}=(4,-1,2)$.

\item \medium\quad
Describe the set of points $\bar{x}\in\mathbb{R}^2$ satisfying $\|\bar{x}\|_1 \le 1$.
Sketch it. How does it compare to the unit disk $\|\bar{x}\|_2 \le 1$?

\item \hard\quad
Using the inner product $\langle f,g\rangle=\int_0^1 f(x)g(x)\,dx$,
prove the Cauchy--Schwarz inequality:
\[
\left(\int_0^1 f(x)g(x)\,dx\right)^2
\le \int_0^1 f(x)^2\,dx\cdot\int_0^1 g(x)^2\,dx.
\]

\end{enumerate}

\section*{Exercise 4: Compute the following improper integrals exactly}
%----------------------------------------------------------

\emph{Where convergent, find the exact value.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $\displaystyle\int_1^{\infty}\frac{dx}{x^2}$

\item \easy\quad $\displaystyle\int_0^{1}\frac{dx}{\sqrt{x}}$

\item \medium\quad $\displaystyle\int_1^{\infty}\frac{\ln x}{x^2}\,dx$
\quad\emph{(integrate by parts)}

\item \medium\quad $\displaystyle\int_0^{\infty} x\,e^{-x}\,dx$

\item \hard\quad $\displaystyle\int_0^{\infty}\frac{dx}{(1+x^2)^2}$
\quad\emph{(use the substitution $x=\tan\theta$)}

\end{enumerate}
