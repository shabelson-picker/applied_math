\section*{Exercise 1: Total differential}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item $z=x^3y-y^3x$: $dz=(3x^2y-y^3)\,dx+(x^3-3xy^2)\,dy$.

\item $z=e^x\sin(xy)$: $z_x=e^x\sin(xy)+e^x y\cos(xy)=e^x(\sin(xy)+y\cos(xy))$,
$z_y=e^x\cdot x\cos(xy)$.
$dz=e^x(\sin(xy)+y\cos(xy))\,dx+xe^x\cos(xy)\,dy$.

\item $u=x^2+y^2+z^2$: $du=2x\,dx+2y\,dy+2z\,dz$.

\item $z=x^2y^3$: $dz=2xy^3\,dx+3x^2y^2\,dy$.

\item $z=\arctan(y/x)$: $z_x=-y/(x^2+y^2)$, $z_y=x/(x^2+y^2)$.
$dz=\frac{-y\,dx+x\,dy}{x^2+y^2}$.

\item $u=xyz+x^2z$: $u_x=yz+2xz$, $u_y=xz$, $u_z=xy+x^2$.
$du=(yz+2xz)\,dx+xz\,dy+(xy+x^2)\,dz$.

\item $z=\ln\!\left(\frac{x+y}{x-y}\right)=\ln(x+y)-\ln(x-y)$.
$z_x=\frac{1}{x+y}-\frac{1}{x-y}=\frac{-2y}{x^2-y^2}$,\quad $z_y=\frac{1}{x+y}+\frac{1}{x-y}=\frac{2x}{x^2-y^2}$.
$dz=\frac{-2y\,dx+2x\,dy}{x^2-y^2}$.

\item $u=\sqrt{x^2+y^2+z^2}=r$: $du=\frac{x\,dx+y\,dy+z\,dz}{\sqrt{x^2+y^2+z^2}}$.

\item $z=x^y=e^{y\ln x}$: $z_x=y x^{y-1}$, $z_y=x^y\ln x$.
$dz=yx^{y-1}\,dx+x^y\ln x\,dy$.

\item $u=f(x^2y,\,y^2z)$. Let $s=x^2y$, $t=y^2z$.
$u_x=f_s\cdot 2xy$, $u_y=f_s\cdot x^2+f_t\cdot 2yz$, $u_z=f_t\cdot y^2$.
$du=2xy f_s\,dx+(x^2 f_s+2yz f_t)\,dy+y^2 f_t\,dz$.

\end{enumerate}

\section*{Exercise 2: Approximation}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item $f(x,y)=x^y$; base point $(a,b)=(1,4)$, $\Delta x=0.02$, $\Delta y=0.05$.
$f(1,4)=1$. $f_x=yx^{y-1}$: at $(1,4)$, $f_x=4$. $f_y=x^y\ln x$: at $(1,4)$, $f_y=0$.
$1.02^{4.05}\approx 1+4(0.02)+0(0.05)=1.08$.

\item $f(x,y)=\ln(x^2+y^2)$; base $(0.09,0.99)\approx(0,1)$, $\Delta x=0.09$, $\Delta y=-0.01$.
Wait, let $a=0,b=1$. But $f(0,1)=\ln 1=0$. $f_x=2x/(x^2+y^2)|_{(0,1)}=0$. $f_y=2y/(x^2+y^2)|_{(0,1)}=2$.
$\ln(0.09^2+0.99^2)\approx 0+0\cdot(0.09)+2\cdot(-0.01)=-0.02$.

\item $f(x,y)=\arctan(y/x)$; base $(a,b)=(1,1)$, $\Delta x=0.05$, $\Delta y=-0.02$.
$f(1,1)=\pi/4$. $f_x=-y/(x^2+y^2)|_{(1,1)}=-1/2$. $f_y=x/(x^2+y^2)|_{(1,1)}=1/2$.
$\arctan(1.05/0.98)\approx\pi/4+(-1/2)(0.05)+(1/2)(-0.02)=\pi/4-0.025-0.01=\pi/4-0.035$.

\item $f(x,y)=x^3 y^2$; base $(1,1)$, $\Delta x=0.02$, $\Delta y=-0.03$.
$f(1,1)=1$. $f_x=3x^2y^2|_{(1,1)}=3$. $f_y=2x^3y|_{(1,1)}=2$.
$1.02^3\cdot 0.97^2\approx 1+3(0.02)+2(-0.03)=1+0.06-0.06=1$.

\item $f(x,y)=\sqrt{5e^x+y^2}$; base $(0,2)$, $\Delta x=0.02$, $\Delta y=0.03$.
$f(0,2)=\sqrt{5+4}=3$. $f_x=\frac{5e^x}{2\sqrt{5e^x+y^2}}|_{(0,2)}=\frac{5}{6}$.
$f_y=\frac{2y}{2\sqrt{5e^x+y^2}}|_{(0,2)}=\frac{4}{6}=\frac{2}{3}$.
$\sqrt{5e^{0.02}+2.03^2}\approx 3+\frac{5}{6}(0.02)+\frac{2}{3}(0.03)=3+\frac{0.1}{6}+0.02=3+0.01\overline{6}+0.02\approx 3.037$.

\item $f(x,y)=\sqrt[4]{x^2+y^2}=(x^2+y^2)^{1/4}$; base $(4,3)$, $\Delta x=0.05$, $\Delta y=-0.07$.
$f(4,3)=(16+9)^{1/4}=25^{1/4}=\sqrt{5}$.
$f_x=\frac{2x}{4(x^2+y^2)^{3/4}}|_{(4,3)}=\frac{8}{4\cdot 25^{3/4}}=\frac{2}{25^{3/4}}$.
$25^{3/4}=(\sqrt{5})^3=5\sqrt{5}$. $f_x=\frac{2}{5\sqrt{5}}$.
Similarly $f_y=\frac{2\cdot 3}{4\cdot 5\sqrt{5}}=\frac{6}{4\cdot 5\sqrt{5}}=\frac{3}{10\sqrt{5}}$.
$\sqrt[4]{4.05^2+2.93^2}\approx\sqrt{5}+\frac{2}{5\sqrt{5}}(0.05)+\frac{3}{10\sqrt{5}}(-0.07)=\sqrt{5}+\frac{0.1}{5\sqrt{5}}-\frac{0.21}{10\sqrt{5}}$.
$=\sqrt{5}+\frac{0.2-0.21}{10\sqrt{5}}=\sqrt{5}-\frac{0.01}{10\sqrt{5}}=\sqrt{5}-\frac{0.001}{\sqrt{5}}$.
$\approx 2.2361-0.000447\approx 2.2357$.

\end{enumerate}

\section*{Exercise 3: Measurement error}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item $S=ab$. $|\Delta S|\approx|b||\Delta a|+|a||\Delta b|=9(0.1)+6(0.1)=0.9+0.6=1.5\,\text{cm}^2$.

\item $V=\pi h(R^2-r^2)$. $V_R=2\pi hR$, $V_r=-2\pi hr$, $V_h=\pi(R^2-r^2)$.
$|\Delta V|\approx 2\pi(7)(5)(0.08)+2\pi(7)(4)(0.08)+\pi(25-16)(0.2)$.
$=2\pi(7)(0.08)(5+4)+\pi(9)(0.2)=2\pi(7)(0.08)(9)+0.2\pi(9)$.
$=2\pi(5.04)+1.8\pi=10.08\pi+1.8\pi=11.88\pi\approx 37.32\,\text{cm}^3$.

\item $\alpha=\arctan(b/a)$. $\alpha_a=-b/(a^2+b^2)$, $\alpha_b=a/(a^2+b^2)$.
At $a=3$, $b=4$: $a^2+b^2=25$. $\alpha_a=-4/25$, $\alpha_b=3/25$.
$|\Delta\alpha|\approx\frac{4}{25}(0.01)+\frac{3}{25}(0.01)=\frac{7}{25}(0.01)=0.0028\,\text{rad}\approx 0.16^{\circ}$.

\item $V=a^2b$. $V_a=2ab$, $V_b=a^2$.
$|\Delta V|\approx 2(20)(10)(0.4)+400(0.2)=160+80=240\,\text{cm}^3$.
$V=400(10)=4000\,\text{cm}^3$.
Relative error: $\frac{|\Delta V|}{V}\approx\frac{240}{4000}=0.06=6\%$.

\end{enumerate}

\section*{Exercise 4: Chain rule}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item $z=u^2\ln v$, $u=y/x$, $v=x^2+y^2$.
$\partial z/\partial x=\partial z/\partial u\cdot\partial u/\partial x+\partial z/\partial v\cdot\partial v/\partial x=2u\ln v\cdot(-y/x^2)+u^2/v\cdot 2x$.
$=2(y/x)\ln(x^2+y^2)\cdot(-y/x^2)+(y^2/x^2)\cdot\frac{2x}{x^2+y^2}=\frac{-2y^2\ln(x^2+y^2)}{x^3}+\frac{2y^2}{x(x^2+y^2)}$.

\item $u=\ln(x+y+z)$, $x=\cos^2 t$, $y=2t\sin t$, $z=t$.
$\frac{du}{dt}=\frac{1}{x+y+z}\left(\frac{dx}{dt}+\frac{dy}{dt}+\frac{dz}{dt}\right)$.
$x'=-2\cos t\sin t=-\sin 2t$, $y'=2\sin t+2t\cos t$, $z'=1$.
$u'=\frac{-\sin 2t+2\sin t+2t\cos t+1}{\cos^2 t+2t\sin t+t}$.

\item $z=2^{x-3y}$, $x=s^2t$, $y=st^2$.
$\partial z/\partial s=\ln 2\cdot 2^{x-3y}(x_s-3y_s)=\ln 2\cdot 2^{s^2t-3st^2}(2st-3t^2)$.
$\partial z/\partial t=\ln 2\cdot 2^{x-3y}(x_t-3y_t)=\ln 2\cdot 2^{s^2t-3st^2}(s^2-6st)$.

\item $w=e^{xy}+x$, $x=u+v$, $y=e^{u+v}$.
$w_u=e^{xy}(y x_u+x y_u)+x_u=e^{xy}(e^{u+v}+x e^{u+v})+1=e^{xy}e^{u+v}(1+x)+1$.
$w_v$ is identical: $w_v=e^{xy}e^{u+v}(1+x)+1$.

\item $F=f(x^2-y^2,e^{xy})$. Let $s=x^2-y^2$, $t=e^{xy}$.
$F_x=f_s\cdot 2x+f_t\cdot ye^{xy}$.
$F_y=f_s\cdot(-2y)+f_t\cdot xe^{xy}$.

\item $u=f(x,y)$, $x=r\cos\theta$, $y=r\sin\theta$.
$u_r=f_x\cos\theta+f_y\sin\theta$, $u_\theta=-f_x r\sin\theta+f_y r\cos\theta$.
Then $u_r^2+\frac{1}{r^2}u_\theta^2=f_x^2\cos^2\theta+2f_xf_y\cos\theta\sin\theta+f_y^2\sin^2\theta+f_x^2\sin^2\theta-2f_xf_y\cos\theta\sin\theta+f_y^2\cos^2\theta=f_x^2+f_y^2$.
So $\left(\frac{\partial u}{\partial x}\right)^2+\left(\frac{\partial u}{\partial y}\right)^2=\left(\frac{\partial u}{\partial r}\right)^2+\frac{1}{r^2}\left(\frac{\partial u}{\partial \theta}\right)^2$.

\item $z=f(x+ay)$. Let $w=x+ay$. Then $z_x=f'(w)\cdot 1=f'(w)$ and $z_y=f'(w)\cdot a=af'(w)=az_x$. $\checkmark$

\item $z=y\varphi(x^2-y^2)$. Let $t=x^2-y^2$.
$z_x=y\varphi'(t)\cdot 2x$. $z_y=\varphi(t)+y\varphi'(t)(-2y)=\varphi(t)-2y^2\varphi'(t)$.
$\frac{z_x}{x}=2y\varphi'(t)$.
$\frac{z_y}{y}=\frac{\varphi(t)}{y}-2y\varphi'(t)$.
$\frac{z_x}{x}+\frac{z_y}{y}=\frac{\varphi(t)}{y}=\frac{z}{y\cdot y}=\frac{z}{y^2}$\quad (since $z=y\varphi(t)$, so $\varphi(t)=z/y$). $\checkmark$

\end{enumerate}

\section*{Exercise 5: Implicit differentiation}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item $F=x^2+y^2+z^2-1$. $F_x=2x$, $F_y=2y$, $F_z=2z$.
$z_x=-F_x/F_z=-x/z$, $z_y=-y/z$.

\item $F=x^2+y^2-z$. $F_x=2x$, $F_y=2y$, $F_z=-1$.
$z_x=2x$, $z_y=2y$.

\item $F=e^{\sin z}-z^2 xy$. $F_x=-z^2 y$, $F_y=-z^2 x$, $F_z=e^{\sin z}\cos z-2xyz$.
$z_x=-(-z^2y)/(e^{\sin z}\cos z-2xyz)=\frac{z^2 y}{e^{\sin z}\cos z-2xyz}$.
$z_y=\frac{z^2 x}{e^{\sin z}\cos z-2xyz}$.

\item $F=xy+yz+xz-1$. $F_x=y+z$, $F_y=x+z$, $F_z=y+x$.
$z_x=-(y+z)/(x+y)$, $z_y=-(x+z)/(x+y)$.

\item $\Phi(cx-az,cy-bz)=0$. Let $u=cx-az$, $v=cy-bz$.
Differentiate w.r.t.\ $x$: $\Phi_u(c-az_x)+\Phi_v(-bz_x)=0\Rightarrow z_x\,(-a\Phi_u-b\Phi_v)=-c\Phi_u$...
Hmm, rather: $\Phi_u(c-az_x)+\Phi_v(0-bz_x)=0\Rightarrow z_x(a\Phi_u+b\Phi_v)=c\Phi_u$.
Differentiate w.r.t.\ $y$: $\Phi_u(-az_y)+\Phi_v(c-bz_y)=0\Rightarrow z_y(a\Phi_u+b\Phi_v)=c\Phi_v$.
$az_x+bz_y=\frac{c(a\Phi_u+b\Phi_v)}{a\Phi_u+b\Phi_v}=c$. $\checkmark$

\item $F=z\ln(x+z)-xy/z$. $F_x=z\cdot\frac{1}{x+z}-y/z$, $F_y=-x/z$,
$F_z=\ln(x+z)+z\cdot\frac{1}{x+z}+xy/z^2$.
$z_x=-F_x/F_z=\dfrac{y/z-z/(x+z)}{\ln(x+z)+1/(x+z)\cdot z+xy/z^2}$
$=\dfrac{y(x+z)-z^2}{z(x+z)\left[\ln(x+z)+z/(x+z)+xy/z^2\right]}$.
$z_y=-F_y/F_z=\dfrac{x/z}{\ln(x+z)+z/(x+z)+xy/z^2}$.

\item $F(x-z,y-z)=0$. Let $s=x-z$, $t=y-z$.
$\frac{\partial}{\partial x}$: $F_s(1-z_x)+F_t(0-z_x)=0\Rightarrow F_s-z_x(F_s+F_t)=0\Rightarrow z_x=\frac{F_s}{F_s+F_t}$.
$\frac{\partial}{\partial y}$: $F_s(-z_y)+F_t(1-z_y)=0\Rightarrow F_t-z_y(F_s+F_t)=0\Rightarrow z_y=\frac{F_t}{F_s+F_t}$.
$z_x+z_y=\frac{F_s+F_t}{F_s+F_t}=1$. $\checkmark$

\end{enumerate}
