\section*{Exercise 1: Find the total differential}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]
\item \easy\quad $z = x^3 y - y^3 x$
\item \easy\quad $z = e^x\sin(xy)$
\item \easy\quad $u = x^2 + y^2 + z^2$
\item \easy\quad $z = x^2 y^3$
\item \medium\quad $z = \arctan\!\left(\dfrac{y}{x}\right)$
\item \medium\quad $u = xyz + x^2 z$
\item \medium\quad $z = \ln\!\left(\dfrac{x+y}{x-y}\right)$
\item \medium\quad $u = \sqrt{x^2 + y^2 + z^2}$
\item \hard\quad $z = x^y$
\item \hard\quad $u = f(x^2 y,\, y^2 z)$, $f$ differentiable.
\end{enumerate}

\section*{Exercise 2: Approximate using the total differential}
%----------------------------------------------------------

\emph{Find a nearby ``nice'' point $(a,b)$, then use
$f(a+\Delta x,b+\Delta y)\approx f(a,b)+f_x\,\Delta x+f_y\,\Delta y$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]
\item \easy\quad $1.02^{4.05}$
\item \easy\quad $\ln(0.09^2+0.99^2)$
\item \easy\quad $\arctan\!\left(\dfrac{1.05}{0.98}\right)$
\item \medium\quad $1.02^3\cdot 0.97^2$
\item \medium\quad $\sqrt{5e^{0.02}+2.03^2}$
\item \hard\quad $\sqrt[4]{4.05^2+2.93^2}$
\end{enumerate}

\section*{Exercise 3: Measurement error analysis}
%----------------------------------------------------------

\emph{Use $|\Delta f|\approx|f_x|\,|\Delta x|+|f_y|\,|\Delta y|$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
Rectangle area $S=ab$, $a=6\,\text{cm}$, $b=9\,\text{cm}$,
precision $\pm 1\,\text{mm}$. Find maximum absolute error in $S$.

\item \medium\quad
Hollow cylinder volume $V=\pi h(R^2-r^2)$,
$R=5\,\text{cm}$, $r=4\,\text{cm}$, $h=7\,\text{cm}$,
precision $\pm 0.8\,\text{mm}$ on radii and $\pm 2\,\text{mm}$ on $h$.

\item \medium\quad
The angle $\alpha$ in a right triangle is computed as $\alpha = \arctan(b/a)$
where $a = 3\,\text{m}$ and $b = 4\,\text{m}$, each measured with error $\pm 0.01\,\text{m}$.
Find the maximum absolute error in $\alpha$.

\item \hard\quad
Rectangular prism $V=a^2 b$, $a=20\,\text{cm}$, $b=10\,\text{cm}$,
precision $\pm 4\,\text{mm}$ and $\pm 2\,\text{mm}$.
Find the maximum \emph{relative} error $|\Delta V|/V$.

\end{enumerate}

\section*{Exercise 4: Chain rule --- find the indicated derivatives}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
$z = u^2\ln v$,\quad $u=y/x$,\quad $v=x^2+y^2$.
Find $\partial z/\partial x$ and $\partial z/\partial y$.

\item \easy\quad
$u=\ln(x+y+z)$,\; $x=\cos^2 t$,\; $y=2t\sin t$,\; $z=t$.
Find $du/dt$.

\item \medium\quad
$z=2^{x-3y}$,\quad $x=s^2 t$,\quad $y=st^2$.
Find $\partial z/\partial s$ and $\partial z/\partial t$.

\item \medium\quad
$w = e^{xy}+x$,\quad $x=u+v$,\quad $y=e^{u+v}$.
Find $\partial w/\partial u$ and $\partial w/\partial v$.

\item \medium\quad
$F(x,y) = f(x^2-y^2,\, e^{xy})$, $f$ differentiable.
Find $\partial F/\partial x$ and $\partial F/\partial y$.

\item \medium\quad
$u = f(x,y)$, $x = r\cos\theta$, $y = r\sin\theta$.
Express $\left(\dfrac{\partial u}{\partial x}\right)^2+\left(\dfrac{\partial u}{\partial y}\right)^2$
in terms of derivatives with respect to $r$ and $\theta$.

\item \hard\quad
$z=f(x+ay)$, $f$ differentiable.
Show that $\dfrac{\partial z}{\partial y}=a\,\dfrac{\partial z}{\partial x}$.

\item \hard\quad
$z = y\,\varphi(x^2-y^2)$, $\varphi$ differentiable. Verify:
\[
  \frac{1}{x}\,\frac{\partial z}{\partial x}+\frac{1}{y}\,\frac{\partial z}{\partial y}=\frac{z}{y^2}.
\]

\end{enumerate}

\section*{Exercise 5: Implicit differentiation}
%----------------------------------------------------------

\emph{Use $\partial z/\partial x = -F_x/F_z$, $\partial z/\partial y = -F_y/F_z$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]
\item \easy\quad $x^2+y^2+z^2=1$. Find $\partial z/\partial x$ and $\partial z/\partial y$.
\item \easy\quad $x^2+y^2-z = 0$. Find $\partial z/\partial x$ and $\partial z/\partial y$.
\item \medium\quad $e^{\sin z}=z^2 xy$. Find $\partial z/\partial x$ and $\partial z/\partial y$.
\item \medium\quad $xy + yz + xz = 1$. Find $\partial z/\partial x$ and $\partial z/\partial y$.
\item \medium\quad $\Phi(cx - az,\; cy - bz)=0$ defines $z$ implicitly. Show:
\[a\,\frac{\partial z}{\partial x}+b\,\frac{\partial z}{\partial y}=c.\]
\item \hard\quad $z\ln(x+z)-xy/z=0$. Find $\partial z/\partial x$ and $\partial z/\partial y$.
\item \hard\quad If $F(x-z, y-z) = 0$ defines $z = z(x,y)$ implicitly, show that
$\dfrac{\partial z}{\partial x}+\dfrac{\partial z}{\partial y}=1$.
\end{enumerate}
