\section*{Exercise 1: Determine convergence and find the sum}
%----------------------------------------------------------

\emph{Use geometric series $\sum_{n=0}^\infty ar^n = a/(1-r)$ for $|r|<1$,
or telescoping where applicable.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $\displaystyle\sum_{n=1}^{\infty}\left(\frac{3}{4}\right)^n$

\item \easy\quad $\displaystyle\sum_{n=0}^{\infty}\frac{2^n+3^n}{5^n}$

\item \easy\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n(n+1)}$
\quad\emph{(telescoping)}

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n(n+2)}$
\quad\emph{(partial fractions)}

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\left[\sin\!\frac{1}{n}-\sin\!\frac{1}{n+1}\right]$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\ln\!\left(1-\frac{1}{n^2}\right)$
\quad\emph{(write as telescoping using $\ln\frac{n-1}{n}+\ln\frac{n+1}{n}$)}

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}\frac{4^{n+1}}{5^n}$

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1}{(3n+1)(3n+4)}$

\end{enumerate}

\section*{Exercise 2: Determine convergence or divergence (no sum needed)}
%----------------------------------------------------------

\emph{State which test you use and why.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^2+1}$

\item \easy\quad $\displaystyle\sum_{n=1}^{\infty}\frac{n}{2^n}$

\item \easy\quad $\displaystyle\sum_{n=1}^{\infty}\frac{n^2}{n!}$

\item \easy\quad $\displaystyle\sum_{n=2}^{\infty}\frac{1}{n(\ln n)^2}$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\frac{2^{1/n}}{3^n}$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\frac{\arctan(1/n)}{n}$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1\cdot 3\cdot 5\cdots(2n-1)}{(2n)!}$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\left(\frac{n}{3n-1}\right)^{2n-1}$

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}\left(\frac{n+1}{2n-1}\right)^n$

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}\frac{n!}{1\cdot 3\cdot 5\cdots(2n-1)}$

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}\frac{(2n)!}{4^n(n!)^2}$

\end{enumerate}

\section*{Exercise 3: Necessary condition \& basic properties}
%----------------------------------------------------------

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad
State the necessary condition for convergence.
Use it to immediately decide on:
$\displaystyle\sum_{n=1}^{\infty}\frac{n^2}{n^2+1}$ and $\displaystyle\sum_{n=1}^{\infty}(-1)^n$.

\item \medium\quad
Prove: if $\sum a_n$ converges and $c>0$, then $\sum ca_n$ converges with sum $c\sum a_n$.

\item \medium\quad
Prove: if $\sum a_n$ converges and $\sum b_n$ diverges, then $\sum(a_n+b_n)$ diverges.

\item \hard\quad
A ball is dropped from height $h=4$ m and bounces to $2/3$ of its previous height.
Find the total distance traveled.

\item \hard\quad
For which values of $x$ does $\displaystyle\sum_{n=1}^{\infty}\tan^n(x)$ converge?
Find the sum where it converges.

\end{enumerate}

\section*{Exercise 4: Integral test}
%----------------------------------------------------------

\emph{The series $\sum f(n)$ and $\int_1^\infty f(x)\,dx$ have the same convergence behaviour
when $f$ is positive, continuous, and decreasing.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $\displaystyle\sum_{n=2}^{\infty}\frac{1}{n\ln n}$

\item \medium\quad $\displaystyle\sum_{n=1}^{\infty}\frac{1}{n^2+1}$
\quad\emph{(use $\int 1/(x^2+1)\,dx = \arctan x$)}

\item \hard\quad $\displaystyle\sum_{n=1}^{\infty}n\,e^{-n^2}$

\end{enumerate}
