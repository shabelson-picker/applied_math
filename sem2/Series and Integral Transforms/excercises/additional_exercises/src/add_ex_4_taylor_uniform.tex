\section*{Exercise 1: Numerical approximation via Taylor polynomials}
%----------------------------------------------------------

\emph{Use the Taylor polynomial $P_n(x) = \sum_{k=0}^n \frac{f^{(k)}(a)}{k!}(x-a)^k$
and the Lagrange remainder $|R_n| \le \frac{M}{(n+1)!}|x-a|^{n+1}$
to achieve the stated accuracy.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad Approximate $e^{0.1}$ with error $< 10^{-4}$.
How many terms of the Maclaurin series are needed?

\item \easy\quad Approximate $\sin(0.2)$ with error $< 10^{-5}$.

\item \easy\quad Approximate $\ln(1.1)$ with error $< 10^{-3}$.

\item \medium\quad Approximate $\sqrt[3]{28}$ using $f(x)=(27+x)^{1/3}$ expanded around $a=0$.
Achieve accuracy $< 0.001$.

\item \medium\quad Approximate $\cos(5^\circ)$ (in radians: $x = \pi/36$) with error $< 10^{-6}$.

\item \medium\quad Using the series for $\arctan x$, approximate $\pi/4$ with error $< 10^{-3}$.
How many terms are needed?

\item \hard\quad Using $\ln 2 = \ln\frac{3}{2} - \ln\frac{3}{4}$ and known series, approximate
$\ln 2$ with error $< 0.001$. Compare efficiency with the direct $\ln(1+x)$ series.

\item \hard\quad Approximate $\displaystyle\int_0^{0.5} e^{-x^2}\,dx$ with error $< 10^{-4}$
by integrating the Maclaurin series term by term.

\end{enumerate}

\section*{Exercise 2: Uniform convergence of sequences of functions}
%----------------------------------------------------------

\emph{A sequence $f_n \to f$ uniformly on $E$ iff $\sup_{x\in E}|f_n(x)-f(x)| \to 0$.
Pointwise convergence alone does NOT imply continuity, integrability, or differentiability of the limit.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad $f_n(x) = x^n$ on $[0,1)$. Find the pointwise limit. Is convergence uniform?

\item \easy\quad $f_n(x) = \dfrac{x}{n}$ on $\mathbb{R}$. Determine pointwise and uniform convergence.

\item \easy\quad $f_n(x) = \dfrac{1}{nx+1}$ on $[1,\infty)$. Show convergence is uniform.

\item \medium\quad $f_n(x) = \dfrac{nx}{1+n^2x^2}$ on $[0,1]$.
Find pointwise limit and determine whether convergence is uniform.

\item \medium\quad $f_n(x) = n^2 x\,e^{-nx}$ on $[0,1]$.
Show $f_n \to 0$ pointwise but $\int_0^1 f_n\,dx \not\to 0$.
What does this tell you about the convergence?

\item \medium\quad $f_n(x) = \dfrac{\sin(nx)}{\sqrt{n}}$ on $[0,2\pi]$.
Show the sequence converges uniformly. Does it converge in $C^1$?

\item \hard\quad $f_n(x) = x^n(1-x)$ on $[0,1]$.
Find the pointwise limit and show convergence is uniform on $[0,1]$.
\quad\emph{(Hint: find $\sup_{x\in[0,1]}|f_n(x)|$ analytically.)}

\item \hard\quad Show that if $f_n \to f$ uniformly on $[a,b]$ and each $f_n$ is continuous,
then $\displaystyle\lim_{n\to\infty}\int_a^b f_n(x)\,dx = \int_a^b f(x)\,dx$.

\end{enumerate}

\section*{Exercise 3: Uniform convergence of series of functions}
%----------------------------------------------------------

\emph{Weierstrass M-test: if $|f_n(x)|\le M_n$ on $E$ and $\sum M_n < \infty$,
then $\sum f_n(x)$ converges uniformly (and absolutely) on $E$.}

\begin{enumerate}[leftmargin=*, label=(\alph*)]

\item \easy\quad Show $\displaystyle\sum_{n=1}^{\infty}\frac{\cos(nx)}{n^2}$ converges
uniformly on $\mathbb{R}$.

\item \easy\quad Show $\displaystyle\sum_{n=0}^{\infty}x^n$ converges uniformly on $[-r,r]$
for any $0<r<1$, but not on $(-1,1)$.

\item \medium\quad Show $\displaystyle\sum_{n=1}^{\infty}\frac{(-1)^n x^{2n}}{n}$
converges uniformly on $[-1,1]$.
\quad\emph{(Use Dirichlet/Abel test for uniform convergence.)}

\item \medium\quad Determine where $\displaystyle\sum_{n=1}^{\infty}\frac{x^n}{n^2+x^2}$
converges uniformly.

\item \medium\quad Let $S(x) = \displaystyle\sum_{n=1}^{\infty}\frac{\sin(nx)}{n^3}$.
Show $S$ is continuous and differentiable, and find $S'(x)$.

\item \hard\quad Let $f(x) = \displaystyle\sum_{n=0}^{\infty}(-1)^n x^{2n}$ for $|x|<1$.
Show the series converges uniformly on $[-r,r]$ for $r<1$,
identify the sum, and verify by differentiating term by term.

\item \hard\quad Show that a power series $\displaystyle\sum_{n=0}^{\infty}a_n x^n$
with radius of convergence $R$ converges uniformly on $[-r,r]$ for any $r<R$.
Why does it generally fail to converge uniformly on $(-R,R)$?

\end{enumerate}
