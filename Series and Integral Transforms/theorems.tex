\section{Numerical Series}

\subsection{Basic Definitions}
\begin{itemize}
  \item The series $\displaystyle\sum_{n=1}^\infty a_n$ \textbf{converges} if the partial sums
        $S_N = \sum_{n=1}^N a_n$ have a finite limit.
  \item \textbf{Necessary condition}: if $\sum a_n$ converges, then $a_n \to 0$.\\
        Contrapositive: if $a_n \not\to 0$, the series diverges.
\end{itemize}

\subsection{Convergence Tests}

\begin{center}
\begin{tabular}{>{\bfseries}lp{10cm}}
\toprule
Test & Statement \\
\midrule
Comparison & If $0 \le a_n \le b_n$ and $\sum b_n$ converges, so does $\sum a_n$. \\[6pt]
Limit Comparison & If $\lim_{n\to\infty} \dfrac{a_n}{b_n} = L \in (0,\infty)$, both series behave the same. \\[6pt]
Ratio (D'Alembert) & $L = \lim \left|\dfrac{a_{n+1}}{a_n}\right|$: converges if $L<1$, diverges if $L>1$, inconclusive if $L=1$. \\[6pt]
Root (Cauchy) & $L = \limsup |a_n|^{1/n}$: converges if $L<1$, diverges if $L>1$. \\[6pt]
Integral & $\sum_{n=1}^\infty f(n)$ and $\int_1^\infty f(x)\,dx$ converge/diverge together\\
         & (for $f$ positive and decreasing). \\[6pt]
Alternating (Leibniz) & $\sum (-1)^n b_n$ converges if $b_n \ge 0$ and $b_n \searrow 0$. \\[6pt]
Dirichlet & $\sum a_n b_n$ converges if partial sums of $a_n$ are bounded and $b_n \searrow 0$. \\[6pt]
Abel & $\sum a_n b_n$ converges if $\sum a_n$ converges and $b_n$ is monotone and bounded. \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Absolute vs.\ Conditional Convergence}
\begin{itemize}
  \item \textbf{Absolutely convergent}: $\sum |a_n| < \infty$ (implies convergent).
  \item \textbf{Conditionally convergent}: $\sum a_n$ converges but $\sum |a_n|$ diverges.
\end{itemize}

\subsection{$p$-Series}
\[
  \sum_{n=1}^\infty \frac{1}{n^p} \quad \text{converges if and only if } p > 1.
\]

\section{Power Series}

\subsection{Radius of Convergence}
The series $\displaystyle\sum_{n=0}^\infty c_n (x-a)^n$ converges absolutely for $|x-a| < R$, where
\[
  R = \frac{1}{\displaystyle\limsup_{n\to\infty} |c_n|^{1/n}}
  \qquad \text{(Cauchy--Hadamard formula)}
\]
Also: $R = \lim_{n\to\infty} \left|\dfrac{c_n}{c_{n+1}}\right|$ when the limit exists.

\subsection{Taylor / Maclaurin Series}
\[
  f(x) = \sum_{n=0}^\infty \frac{f^{(n)}(a)}{n!}(x-a)^n
\]

\subsection{Common Maclaurin Series}
\begin{align*}
  e^x &= \sum_{n=0}^\infty \frac{x^n}{n!}, \qquad R = \infty \\[6pt]
  \sin x &= \sum_{n=0}^\infty \frac{(-1)^n\, x^{2n+1}}{(2n+1)!}, \qquad R = \infty \\[6pt]
  \cos x &= \sum_{n=0}^\infty \frac{(-1)^n\, x^{2n}}{(2n)!}, \qquad R = \infty \\[6pt]
  \frac{1}{1-x} &= \sum_{n=0}^\infty x^n, \qquad |x| < 1 \\[6pt]
  \ln(1+x) &= \sum_{n=1}^\infty \frac{(-1)^{n+1}\, x^n}{n}, \qquad |x| \le 1,\ x \ne -1 \\[6pt]
  \arctan x &= \sum_{n=0}^\infty \frac{(-1)^n\, x^{2n+1}}{2n+1}, \qquad |x| \le 1 \\[6pt]
  (1+x)^\alpha &= \sum_{n=0}^\infty \binom{\alpha}{n} x^n, \qquad |x| < 1
\end{align*}

\subsection{Taylor's Theorem with Remainder}

\begin{theorem}[Taylor's Theorem --- Lagrange Remainder]
If $f$ has $n+1$ continuous derivatives on an interval containing $a$, then
\[
  f(x) = \sum_{k=0}^n \frac{f^{(k)}(a)}{k!}(x-a)^k + R_n(x),
\]
where the \textbf{Lagrange remainder} is
\[
  R_n(x) = \frac{f^{(n+1)}(c)}{(n+1)!}(x-a)^{n+1}
\]
for some $c$ strictly between $a$ and $x$.
\end{theorem}

\textbf{Error bound:}
\[
  |R_n(x)| \leq \frac{M_{n+1}}{(n+1)!}\,|x-a|^{n+1},
  \qquad M_{n+1} = \sup_{t \text{ between } a \text{ and } x}\bigl|f^{(n+1)}(t)\bigr|.
\]

\textbf{Workflow for ``how many terms'' questions:}
Bound $|R_n(x)|$ using $M_{n+1}$, then solve $\frac{M_{n+1}}{(n+1)!}|x-a|^{n+1} < \varepsilon$ for the smallest $n$.

\subsection{Computing Limits via Taylor Series}

To evaluate $\lim_{x\to 0} g(x)$:
\begin{enumerate}
  \item Expand numerator and denominator as power series around 0.
  \item Cancel common factors of $x^k$.
  \item Read off the limit from the leading terms.
\end{enumerate}
For $\lim_{x\to 0} \frac{d^k}{dx^k}\bigl[f(x)\bigr]$: write $f(x) = \sum a_n x^n$, then $f^{(k)}(0) = k!\,a_k$.

\section{Uniform Convergence}

\subsection{Sequences of Functions}

\begin{definition}
Let $f_n, f : D \to \mathbb{R}$.
\begin{itemize}
  \item \textbf{Pointwise convergence}: $f_n \to f$ if for each $x \in D$, $f_n(x) \to f(x)$.
  \item \textbf{Uniform convergence} ($f_n \rightrightarrows f$):
    \[
      \sup_{x \in D}|f_n(x) - f(x)| \;\to\; 0.
    \]
\end{itemize}
Uniform $\Rightarrow$ pointwise, but pointwise $\not\Rightarrow$ uniform in general.
\end{definition}

\textbf{Standard method:} Compute $M_n = \sup_{x\in D}|f_n(x)-f(x)|$ (find the max by differentiation or AM-GM). If $M_n \to 0$: uniform; if $M_n \not\to 0$: not uniform.

\textbf{Negation:} $f_n \not\rightrightarrows f$ iff $\exists\,\varepsilon > 0$ and a sequence $x_n \in D$ with $|f_n(x_n) - f(x_n)| \geq \varepsilon$.
A typical choice is $x_n = \arg\max_x |f_n(x) - f(x)|$.

\begin{example}[$f_n(x) = x^n$ on $[0,1)$]
Pointwise limit: $f(x) = 0$. But $\sup_{x\in[0,1)}|x^n - 0| = 1 \not\to 0$. Not uniform.
On $[0,r]$ for any $r < 1$: $\sup = r^n \to 0$. Uniform.
\end{example}

\subsection{Properties Preserved by Uniform Convergence}

\begin{theorem}[Continuity]
If each $f_n$ is continuous on $D$ and $f_n \rightrightarrows f$, then $f$ is continuous on $D$.
\end{theorem}

\begin{theorem}[Term-by-term Integration]
If $f_n \rightrightarrows f$ on $[a,b]$, then
\[
  \lim_{n\to\infty}\int_a^b f_n(x)\,dx = \int_a^b f(x)\,dx.
\]
\end{theorem}

\begin{theorem}[Term-by-term Differentiation]
If $f_n \to f$ pointwise on $[a,b]$, each $f_n$ is $C^1$, and $f_n' \rightrightarrows g$ uniformly, then $f$ is differentiable and $f' = g$.
\end{theorem}

\begin{remark}
Counterexample for pointwise-only: $f_n(x) = n^2 x\,e^{-nx}$ on $[0,1]$ converges pointwise to $0$, but $\int_0^1 f_n\,dx = 1 \not\to 0$. The convergence is not uniform.
\end{remark}

\subsection{Series of Functions}

A series $\displaystyle\sum_{n=1}^\infty u_n(x)$ \textbf{converges uniformly} on $D$ if its partial sums
$S_N(x) = \sum_{n=1}^N u_n(x)$ converge uniformly on $D$.

\begin{theorem}[Weierstrass $M$-Test]
If $|u_n(x)| \leq M_n$ for all $x \in D$ and $\displaystyle\sum_{n=1}^\infty M_n < \infty$, then
$\sum u_n(x)$ converges \emph{uniformly and absolutely} on $D$.
\end{theorem}

\begin{theorem}[Dirichlet's Test --- Uniform Version]
$\sum a_n(x)\,b_n(x)$ converges uniformly on $D$ if:
\begin{enumerate}
  \item The partial sums $A_N(x) = \sum_{n=1}^N a_n(x)$ are \emph{uniformly bounded} on $D$.
  \item $b_n(x) \searrow 0$ monotonically and \emph{uniformly} on $D$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Abel's Test --- Uniform Version]
$\sum a_n(x)\,b_n(x)$ converges uniformly on $D$ if:
\begin{enumerate}
  \item $\sum a_n(x)$ converges \emph{uniformly} on $D$.
  \item $b_n(x)$ is monotone and \emph{uniformly bounded} on $D$.
\end{enumerate}
\end{theorem}

\textbf{Key fact about power series:} $\displaystyle\sum_{n=0}^\infty c_n x^n$ with radius of convergence $R$ converges
uniformly on $[-r,r]$ for any $r < R$, but generally \emph{not} uniformly on the open interval $(-R,R)$.

\textbf{Properties} (same as for sequences): if $\sum u_n$ converges uniformly and each $u_n$ is continuous, then the sum is continuous; term-by-term integration and differentiation are valid under uniform convergence (for differentiation: need $\sum u_n'$ to converge uniformly).

\section{Fourier Series}

\subsection{Definition}
For $f$ periodic with period $2L$:
\[
  f(x) \sim \frac{a_0}{2} + \sum_{n=1}^\infty
  \left( a_n \cos\frac{n\pi x}{L} + b_n \sin\frac{n\pi x}{L} \right)
\]
\textbf{Coefficients}:
\[
  a_n = \frac{1}{L}\int_{-L}^{L} f(x)\cos\frac{n\pi x}{L}\,dx,
  \qquad
  b_n = \frac{1}{L}\int_{-L}^{L} f(x)\sin\frac{n\pi x}{L}\,dx
\]

\subsection{Dirichlet Conditions}
If $f$ is piecewise smooth and periodic, the Fourier series converges to:
\begin{itemize}
  \item $f(x)$ at points of continuity,
  \item $\dfrac{f(x^+)+f(x^-)}{2}$ at jump discontinuities.
\end{itemize}

\subsection{Even and Odd Functions}
\begin{itemize}
  \item \textbf{Even} $f$: $b_n = 0$ (cosine series only).
  \item \textbf{Odd} $f$: $a_n = 0$ (sine series only).
\end{itemize}

\subsection{Parseval's Identity}
\[
  \frac{1}{L}\int_{-L}^{L}|f(x)|^2\,dx
  = \frac{a_0^2}{2} + \sum_{n=1}^\infty \left(a_n^2 + b_n^2\right)
\]

\subsection{Complex Form}
\[
  f(x) = \sum_{n=-\infty}^\infty c_n\, e^{in\pi x/L},
  \qquad
  c_n = \frac{1}{2L}\int_{-L}^{L} f(x)\,e^{-in\pi x/L}\,dx
\]

\section{Fourier Transform}

\subsection{Definition}
\[
  \hat{f}(\xi) = \mathcal{F}\{f\}(\xi)
  = \int_{-\infty}^\infty f(x)\,e^{-2\pi i\xi x}\,dx
\]
(Alternative convention: $\hat{f}(\omega) = \int_{-\infty}^\infty f(x)\,e^{-i\omega x}\,dx$)

\subsection{Key Properties}

\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{Property} & \textbf{Formula} \\
\midrule
Linearity   & $\mathcal{F}\{af+bg\} = a\hat{f}+b\hat{g}$ \\[6pt]
Shift       & $\mathcal{F}\{f(x-a)\} = e^{-2\pi i a\xi}\,\hat{f}(\xi)$ \\[6pt]
Scaling     & $\mathcal{F}\{f(ax)\} = \dfrac{1}{|a|}\,\hat{f}(\xi/a)$ \\[6pt]
Derivative  & $\mathcal{F}\{f'\}(\xi) = 2\pi i\xi\,\hat{f}(\xi)$ \\[6pt]
Convolution & $\mathcal{F}\{f*g\} = \hat{f}\cdot\hat{g}$ \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Parseval / Plancherel Theorem}
\[
  \int_{-\infty}^\infty |f(x)|^2\,dx = \int_{-\infty}^\infty |\hat{f}(\xi)|^2\,d\xi
\]

\section{Laplace Transform}

\subsection{Definition}
\[
  \mathcal{L}\{f\}(s) = F(s) = \int_0^\infty f(t)\,e^{-st}\,dt
\]

\subsection{Common Transforms}

\begin{center}
\begin{tabular}{ll}
\toprule
$f(t)$ & $F(s)$ \\
\midrule
$1$          & $\dfrac{1}{s}$ \\[8pt]
$t^n$        & $\dfrac{n!}{s^{n+1}}$ \\[8pt]
$e^{at}$     & $\dfrac{1}{s-a}$ \\[8pt]
$\sin(bt)$   & $\dfrac{b}{s^2+b^2}$ \\[8pt]
$\cos(bt)$   & $\dfrac{s}{s^2+b^2}$ \\[8pt]
$e^{at}f(t)$ & $F(s-a)$ \\[8pt]
$f'(t)$      & $sF(s)-f(0)$ \\[8pt]
$f''(t)$     & $s^2F(s)-sf(0)-f'(0)$ \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Convolution Theorem}
\[
  \mathcal{L}\{f * g\} = F(s) \cdot G(s)
\]
